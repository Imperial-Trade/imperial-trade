================================================================================
RUN THESE COMMANDS DIRECTLY ON THE VPS
================================================================================

STEP 1: Check current status
-----------------------------
cd C:\vps-broker-service
pm2 status
Get-NetTCPConnection -LocalPort 3001 -ErrorAction SilentlyContinue


STEP 2: Check if Python scripts are updated
--------------------------------------------
Get-Content python\test_connection.py | Select-String "Program Files"

If this shows "Program Files", scripts are REVERTED.
If it shows "MT5_BrokerService", scripts still need updating.


STEP 3: Rebuild and restart (if needed)
-----------------------------------------
cd C:\vps-broker-service
npm run build
pm2 restart imperial-trade-broker-service
Start-Sleep -Seconds 3
pm2 status


STEP 4: Verify everything
--------------------------
Write-Host "=== VERIFICATION ===" -ForegroundColor Cyan

# Check PM2
pm2 status

# Check port
$p = Get-NetTCPConnection -LocalPort 3001 -ErrorAction SilentlyContinue
if ($p) { Write-Host "✅ Port 3001: LISTENING" -ForegroundColor Green } else { Write-Host "❌ Port 3001: NOT LISTENING" -ForegroundColor Red }

# Check Python path
$t = Get-Content C:\vps-broker-service\python\test_connection.py -Raw
if ($t -match "C:\\Program Files\\MetaTrader 5") { Write-Host "✅ Python: REVERTED" -ForegroundColor Green } else { Write-Host "❌ Python: NOT REVERTED" -ForegroundColor Red }

# Check MT5
$m = Get-Process -Name terminal64 -ErrorAction SilentlyContinue
if ($m) { Write-Host "✅ MT5: Running" -ForegroundColor Green } else { Write-Host "❌ MT5: Not Running" -ForegroundColor Red }
